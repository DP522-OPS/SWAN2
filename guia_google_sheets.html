Configuracion Google Sheets + Tablero Maestro

Siga estos pasos UNA sola vez. Despues la app envia datos automaticamente.

1

Crear una nueva hoja de Google Sheets

Abra el navegador y vaya a:

sheets.google.com

Haga clic en "+ En blanco" para crear una hoja nueva.

Ponle nombre: Swantex Produccion

Guarde la URL — la necesita en el paso 6. Tiene esta forma:
https://docs.google.com/spreadsheets/d/ESTE_ES_EL_ID/edit
2

Abrir el editor de Apps Script

Desde la hoja de Sheets, haga clic en el menu:

Extensiones → Apps Script

Se abre una nueva ventana con un editor de codigo.

3

Pegar el codigo del servidor

Borre todo el contenido que aparece en el editor (Ctrl+A, Delete) y pegue este codigo:

// ============================================================
// SWANTEX - Google Apps Script

var HOJA = 'Produccion';
var HOJA_ESTADO = 'EnCurso';

function doGet(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var accion = (e.parameter && e.parameter.action) ? e.parameter.action : 'leer';
    var resultado;
    if (accion === 'escribir') {
      var datos = JSON.parse(decodeURIComponent(e.parameter.d || '{}'));
      resultado = escribirFila(datos);
    } else if (accion === 'estado_actual') {
      var datosEstado = JSON.parse(decodeURIComponent(e.parameter.d || '{}'));
      resultado = actualizarEstado(datosEstado);
    } else if (accion === 'leer_estado') {
      resultado = leerEstadoActual();
    } else {
      var fecha = e.parameter.fecha || '';
      resultado = leerFilas(fecha);
    }
    return ContentService
      .createTextOutput(JSON.stringify(resultado))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({error: err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

// Guarda/actualiza, por maquina, que referencia esta corriendo AHORA MISMO.
// Una fila por MaqID (se sobreescribe cada vez que se abre o cierra un lote
// en esa maquina). Esto es lo que permite ver "lo que esta corriendo" sin
// esperar a que se cierre el lote.
function actualizarEstado(d) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(HOJA_ESTADO);
  var headers = ['MaqID','RefID','Referencia','Fecha','HoraInicio','Estado','Timestamp'];
  if (!sh) {
    sh = ss.insertSheet(HOJA_ESTADO);
    sh.appendRow(headers);
    sh.getRange(1,1,1,headers.length).setFontWeight('bold').setBackground('#1e3a5f').setFontColor('#ffffff');
    sh.setFrozenRows(1);
  }
  var rowVals = [
    d.maqId || '', d.refId || '', d.refNombre || '',
    d.fecha || '', d.hi || '', d.estado || '', new Date().toISOString()
  ];
  var lastRow = sh.getLastRow();
  var rowIdx = -1;
  if (lastRow > 1) {
    var maqCol = sh.getRange(2, 1, lastRow - 1, 1).getValues();
    for (var i = 0; i < maqCol.length; i++) {
      if (maqCol[i][0] === d.maqId) { rowIdx = i + 2; break; }
    }
  }
  if (rowIdx > 0) {
    sh.getRange(rowIdx, 1, 1, rowVals.length).setValues([rowVals]);
  } else {
    sh.appendRow(rowVals);
  }
  return { ok: true };
}

function leerEstadoActual() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(HOJA_ESTADO);
  if (!sh || sh.getLastRow() <= 1) return { estados: [] };
  var datos = sh.getDataRange().getValues();
  var cabeceras = datos[0];
  var estados = [];
  for (var i = 1; i < datos.length; i++) {
    var fila = {};
    for (var j = 0; j < cabeceras.length; j++) {
      fila[cabeceras[j]] = datos[i][j];
    }
    estados.push(fila);
  }
  return { estados: estados };
}

function escribirFila(d) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(HOJA);
  var headers = [
    'Timestamp','Fecha','Equipo','MaqID','Modelo','Diametro',
    'OperarioID','Operario','Turno','RefID','Referencia','Talla',
    'LotePlaneado','CtrInicio','CtrFin','Tubos',
    'HoraInicio','HoraFin','Duracion','ParosMin',
    'StdMin','Eficiencia','Observaciones','ParosDetalle'
  ];
  if (!sh) {
    sh = ss.insertSheet(HOJA);
    sh.appendRow(headers);
    sh.getRange(1,1,1,headers.length).setFontWeight('bold').setBackground('#1e3a5f').setFontColor('#ffffff');
    sh.setFrozenRows(1);
  } else {
    // Migracion automatica: agrega la columna ParosDetalle si la hoja es de antes
    var lastCol = sh.getLastColumn();
    var headerRow = lastCol ? sh.getRange(1,1,1,lastCol).getValues()[0] : [];
    if (headerRow.indexOf('ParosDetalle') === -1) {
      sh.getRange(1, lastCol+1).setValue('ParosDetalle')
        .setFontWeight('bold').setBackground('#1e3a5f').setFontColor('#ffffff');
    }
  }
  sh.appendRow([
    new Date().toISOString(),
    d.fecha || '', d.equipo || '', d.maqId || '', d.modelo || '', d.diam || '',
    d.opId || '', d.opNombre || '', d.turno || '',
    d.refId || '', d.refNombre || '', d.talla || '',
    d.lotSize || 0, d.ctrStart !== null ? d.ctrStart : '',
    d.ctrEnd !== null ? d.ctrEnd : '', d.tubos || 0,
    d.hi || '', d.hf || '', d.dur || '', d.parosMin || 0,
    d.std || '', d.eff !== null ? d.eff : '', d.obs || '',
    d.parosDetalle || ''
  ]);
  return { ok: true, fila: sh.getLastRow() };
}

function leerFilas(fechaFiltro) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(HOJA);
  if (!sh || sh.getLastRow() <= 1) return { filas: [] };
  var datos = sh.getDataRange().getValues();
  var cabeceras = datos[0];
  var filas = [];
  for (var i = 1; i < datos.length; i++) {
    var fila = {};
    for (var j = 0; j < cabeceras.length; j++) {
      fila[cabeceras[j]] = datos[i][j];
    }
    if (!fechaFiltro || fila['Fecha'] === fechaFiltro) {
      filas.push(fila);
    }
  }
  return { filas: filas, total: filas.length };
}

// "Recibo de hilo" de hilos.html). La clave NUNCA va en la pagina:

//   QWEN_BASE_URL  (opcional)     por defecto https://dashscope-intl.aliyuncs.com/compatible-mode/v1
//   QWEN_MODEL     (opcional)     por defecto qwen-vl-max
//   MAX_FOTOS_DIA  (opcional)     tope de fotos por dia, por defecto 300
// ============================================================
var PROMPT_ROTULO = [
  'You read photos of yarn box labels from a textile plant in Colombia.',
  'Two kinds of label exist: (A) printed supplier labels (often Chinese/English, or Hyosung/Creora), and',
  '(B) handwritten Swantex labels "F-PRH-005 IDENTIFICACION DE MATERIALES" with the fields Liberacion, Item, Nro. (top right),',
  'Fecha, Maquina, Material, Proveedor, Titulo, Torsion, Lote, Unidades, Peso bruto, Tara, Peso neto.',
  'Return ONLY one JSON object, no other text, with exactly these keys:',
  '{"tipo_rotulo":"proveedor|swantex|desconocido","lote":string|null,"especificacion":string|null,"titulo":string|null,',
  '"unidades":number|null,"peso_neto":number|null,"peso_bruto":number|null,"tara":number|null,',
  '"torsion":"S"|"Z"|null,"caja":string|null,"dudas":[string]}.',
  'Meaning: lote = LOT NO / Lot Number / Lote, copied exactly as printed or written (do not add or remove characters; Swantex lots look like OS-954).',
  'especificacion = SPEC / Denier (e.g. 40D/12F). titulo = the Titulo field of a Swantex label (e.g. 17-8-5).',
  'unidades = number of cones/bobbins (BOBBIN, Units, Unidades). peso_neto = N.W. / Net Wt / Peso neto in kg.',
  'peso_bruto = G.W. / Gross Wt / Peso bruto in kg. tara = Tara in kg.',
  'torsion = S or Z only if written on the label. caja = the number written under Nro. on Swantex labels (null on supplier labels).',
  'Rules: NEVER guess. If a value is blurry, covered by a stamp, cut off or ambiguous, use null and add its key name to dudas.',
  'Numbers use a dot as decimal separator and no units. Ignore green or purple ink stamps and dates stamped by the plant.'
].join(' ');

function doPost(e) {
  var salida;
  try {
    var body = JSON.parse((e.postData ? e.postData.contents : '') || '{}');
    if (body.action === 'leer_rotulo') {
      salida = leerRotulo(body.imagen);
    } else {
      salida = { error: 'Accion desconocida' };
    }
  } catch (err) {
    salida = { error: err.message };
  }
  return ContentService.createTextOutput(JSON.stringify(salida)).setMimeType(ContentService.MimeType.JSON);
}

function leerRotulo(imagen) {
  var props = PropertiesService.getScriptProperties();
  var key = props.getProperty('QWEN_API_KEY');
  if (!key) return { error: 'Falta QWEN_API_KEY en las propiedades del script.' };
  if (!imagen || String(imagen).indexOf('data:image/') !== 0) return { error: 'Imagen invalida.' };
  if (imagen.length > 7000000) return { error: 'Imagen demasiado grande.' };

  // Tope diario para que nadie gaste credito por error
  var max = parseInt(props.getProperty('MAX_FOTOS_DIA') || '300', 10);
  var hoy = Utilities.formatDate(new Date(), 'America/Bogota', 'yyyy-MM-dd');
  var cont = (props.getProperty('FOTOS_CONTADOR') || '').split('|');
  var n = (cont[0] === hoy) ? parseInt(cont[1] || '0', 10) : 0;
  if (n >= max) return { error: 'Limite diario de fotos alcanzado (' + max + ').' };
  props.setProperty('FOTOS_CONTADOR', hoy + '|' + (n + 1));

  var base = (props.getProperty('QWEN_BASE_URL') || 'https://dashscope-intl.aliyuncs.com/compatible-mode/v1').replace(/\/+$/, '');
  var modelo = props.getProperty('QWEN_MODEL') || 'qwen-vl-max';
  var resp = UrlFetchApp.fetch(base + '/chat/completions', {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + key },
    muteHttpExceptions: true,
    payload: JSON.stringify({
      model: modelo,
      temperature: 0,
      messages: [{ role: 'user', content: [
        { type: 'image_url', image_url: { url: imagen } },
        { type: 'text', text: PROMPT_ROTULO }
      ] }]
    })
  });
  var code = resp.getResponseCode();
  var txt = resp.getContentText();
  if (code !== 200) return { error: 'Qwen respondio ' + code + ': ' + txt.substring(0, 300) };
  var j = JSON.parse(txt);
  var out = j.choices[0].message.content;
  if (Array.isArray(out)) {
    out = out.map(function (p) { return p.text || ''; }).join('');
  }
  var a = out.indexOf('{');
  var z = out.lastIndexOf('}');
  if (a === -1 || z === -1) return { error: 'Qwen no devolvio datos legibles: ' + out.substring(0, 200) };
  return { ok: true, datos: JSON.parse(out.substring(a, z + 1)), modelo: modelo };
}

// Ejecutela UNA vez desde el editor (boton Ejecutar): pide los permisos
// de conexion externa y comprueba que la clave, la URL y el modelo funcionan.
function probarQwen() {
  var props = PropertiesService.getScriptProperties();
  var base = (props.getProperty('QWEN_BASE_URL') || 'https://dashscope-intl.aliyuncs.com/compatible-mode/v1').replace(/\/+$/, '');
  var resp = UrlFetchApp.fetch(base + '/chat/completions', {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + props.getProperty('QWEN_API_KEY') },
    muteHttpExceptions: true,
    payload: JSON.stringify({
      model: props.getProperty('QWEN_MODEL') || 'qwen-vl-max',
      messages: [{ role: 'user', content: 'Responde solo: OK' }]
    })
  });
  Logger.log(resp.getResponseCode() + ' ' + resp.getContentText().substring(0, 300));
}

4

Desplegar como aplicacion web

En el editor de Apps Script, haga clic en:

"Implementar" → "Nueva implementacion"

Configure asi:

• Tipo: Aplicacion web

• Ejecutar como: Yo (su cuenta de Google)

• Quien tiene acceso: Cualquier persona

Haga clic en "Implementar". Puede pedir autorizar permisos — acepte todo.

Si ya tenia una implementacion antigua y solo actualizo el codigo (por ejemplo para agregar ParosDetalle), NO cree una implementacion nueva porque cambia la URL. En cambio vaya a "Implementar" → "Administrar implementaciones", haga clic en el lapiz (editar) de la implementacion activa, elija "Version: Nueva version" y de clic en "Implementar". Asi la URL sigue siendo la misma y no hay que volver a pegarla en la app ni en el tablero.
5

Copiar la URL del servidor

Despues de implementar aparece una URL de aplicacion web. Tiene esta forma:

https://script.google.com/macros/s/XXXXXXXXXXXX/exec

Copie esa URL — la necesita en el siguiente paso.

Con esa URL el servidor esta listo. No necesita tocar nada mas en Apps Script.
6

Pegar la URL en la app de operarios

Abra la app Swantex (dp522-ops.github.io/SWAN2)

Vaya a Config → Nube

Pegue la URL en el campo "URL del Servidor" y guarde.

Desde ese momento cada lote que se guarda se envia automaticamente a Google Sheets.

7

Abrir el tablero maestro

Abra el archivo dashboard.html (que tambien subio a GitHub)

URL: dp522-ops.github.io/SWAN2/dashboard.html

Pegue la misma URL del servidor y vera la produccion en tiempo real.

El tablero se actualiza automaticamente cada 30 segundos. Puede dejarlo abierto en el PC o en una TV.
8

Materia prima (hilos) en tiempo real

Abra hilos.html y pegue ahi la misma URL del servidor (paso 5).

Con el codigo actualizado (paso 3), cada vez que un operario abre o cierra un lote la app avisa al servidor que referencia esta corriendo en cada maquina. La pestana "Lo que esta corriendo" de Hilos usa eso para calcular cuanto hilo se necesita hoy, sin esperar a que se cierren los lotes.

9

Lectura de rotulos de cajas (pestana "Recibo de hilo")

Con el codigo del paso 3 ya actualizado (incluye doPost), falta guardar la clave de Qwen en el servidor:

1) En Apps Script, engranaje Configuracion del proyecto → Propiedades del script → Agregar propiedad: QWEN_API_KEY = su clave de Alibaba Cloud Model Studio. Opcionales: QWEN_BASE_URL (copiela tal cual la muestra la consola de Qwen para su region) y QWEN_MODEL (por defecto qwen-vl-max).

2) En el editor elija la funcion probarQwen y pulse Ejecutar. Acepte los permisos. En Registro de ejecucion debe aparecer "200 ...".

3) Implementar → Administrar implementaciones → lapiz → Version: Nueva version → Implementar (la URL no cambia).

4) En hilos.html, pestana Recibo de hilo, pulse Probar conexion con Qwen: si todo esta bien dice "Conexion OK" y, si no, indica que revisar.

La clave se guarda solo en el servidor, nunca en la pagina de GitHub (que es publica). El servidor corta a 300 fotos por dia por si alguien lo llama por error (cambie MAX_FOTOS_DIA si necesita mas).
